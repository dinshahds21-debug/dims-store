function handleRegister() { const userBaru = document.getElementById('reg-user').value; const passBaru = document.getElementById('reg-pass').value; if (!userBaru || !passBaru) { alert("Isi semua data!"); return; } // 1. Ambil daftar user yang sudah ada, atau buat array kosong jika belum ada let users = JSON.parse(localStorage.getItem('allUsers')) || []; // 2. Cek apakah username sudah dipakai const sudahAda = users.find(u => u.username === userBaru); if (sudahAda) { alert("Username sudah terdaftar!"); return; } // 3. Masukkan user baru ke array users.push({ username: userBaru, password: passBaru }); // 4. Simpan kembali ke LocalStorage localStorage.setItem('allUsers', JSON.stringify(users)); alert("Registrasi Berhasil! Silakan Login."); window.location.href = "login.html"; }
